Eingaben umwandeln

Console.Write("Zahl: ");
string? input = Console.ReadLine();

if (int.TryParse(input, out int number)) {     // sicher: kein Absturz bei Unsinn
    Console.WriteLine($"Das Doppelte: {number * 2}");
} else {
    Console.WriteLine("Keine gültige Zahl");
}

int direct = int.Parse("42");                  // wirft Exception bei ungültigem Text
