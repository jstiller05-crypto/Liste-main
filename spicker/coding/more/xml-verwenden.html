XML verwenden

Mit XPath gezielt Knoten finden, dann je Sprache einlesen – und Parser-Fehler immer abfangen.

XPath-Grundlagen

XPath ist eine eigene kleine Sprache, um Knoten in einem XML-Dokument zu finden – für das library-Beispiel:

XPathBedeutet
/library/bookjedes book direkt unter library
//bookjedes book, egal wo im Dokument
@iddas Attribut id des aktuellen Elements
book[1]das erste book (1-basiert, nicht 0!)
book[@lang='de']nur book-Elemente mit Attribut lang="de"
text()der reine Textinhalt eines Elements

JavaScript

const xmlText = "<library><book id=\"1\"><title>Clean Code</title></book></library>";

const parser = new DOMParser();
const doc = parser.parseFromString(xmlText, "application/xml");

// Bei einem Syntaxfehler baut der Browser statt eines echten Dokuments
// ein -Element ein -> das muss man selbst abfragen
const parserError = doc.querySelector("parsererror");
if (parserError) {
  console.error("[xml] Parser-Fehler:", parserError.textContent);
} else {
  const title = doc.querySelector("book title").textContent;
  console.log("[xml] Titel:", title);

  // getElementsByTagName funktioniert genauso wie im HTML-DOM
  const books = doc.getElementsByTagName("book");
  console.log("[xml] Anzahl Bücher:", books.length);
}

// Umgekehrte Richtung: XML-Dokument zurück in Text
const serializer = new XMLSerializer();
console.log(serializer.serializeToString(doc));

PHP (SimpleXML)

<?php
// Eigene Fehlerbehandlung statt PHP-Warnungen -> gibt false statt eines Objekts zurück
libxml_use_internal_errors(true);

$xml = simplexml_load_file("library.xml");

if ($xml === false) {
    foreach (libxml_get_errors() as $error) {
        error_log("[xml] Fehler: " . $error->message);
    }
} else {
    foreach ($xml->book as $book) {
        echo $book->title . " (id=" . $book['id'] . ")\n";  // Attribut per ['id']
    }
}

C++ (Qt)

#include <QXmlStreamReader>
#include <QFile>
#include <QDebug>

QFile file("library.xml");
if (!file.open(QIODevice::ReadOnly)) {
    qWarning() << "[xml] Datei konnte nicht geoeffnet werden:" << file.errorString();
    return;
}

QXmlStreamReader reader(&file);
while (!reader.atEnd()) {
    reader.readNext();
    if (reader.isStartElement() && reader.name() == QLatin1String("book")) {
        // attributes() liefert alle Attribute des AKTUELLEN Start-Tags
        QString id = reader.attributes().value("id").toString();
        qDebug() << "book id:" << id;
    }
}

if (reader.hasError()) {
    // hasError() erst NACH der Schleife prüfen - haeufigster Fehler hier
    qWarning() << "[xml] Parser-Fehler:" << reader.errorString();
}
