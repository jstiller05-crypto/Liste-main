Klassen & Objekte

Objektorientierte Programmierung in PHP – mit Konstruktor, Sichtbarkeit und Vererbung.

Klasse

<?php
class Player {
    // Constructor Promotion (PHP 8): Eigenschaften direkt im Konstruktor anlegen
    public function __construct(
        private string $name,
        private int $score = 0
    ) {}

    public function addScore(int $points): void {
        if ($points < 0) {
            throw new InvalidArgumentException("Punkte dürfen nicht negativ sein");
        }
        $this->score += $points;       // $this-> für Zugriff auf Eigenschaften
    }

    public function getName(): string {
        return $this->name;
    }

    public function getScore(): int {
        return $this->score;
    }
}

$player = new Player("Max");
$player->addScore(50);                 // -> ruft Methoden auf
echo $player->getName() . ": " . $player->getScore();

Vererbung

class Admin extends Player {
    public function getName(): string {
        return "[Admin] " . parent::getName();   // Methode der Elternklasse aufrufen
    }
}
