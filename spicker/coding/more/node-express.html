Express

Das beliebteste Framework für Webserver und APIs in Node.js.

Installieren

npm init -y
npm install express

Server mit Routen

import express from "express";

const app = express();
const PORT = 3000;

app.use(express.json());                 // JSON im Request-Body lesen
app.use(express.static("public"));       // Dateien aus /public ausliefern (HTML, CSS, JS, Bilder)

const tasks = [
  { id: 1, title: "HTML lernen", done: true },
  { id: 2, title: "Node.js lernen", done: false }
];

// alle Aufgaben
app.get("/api/tasks", (req, res) => {
  res.json(tasks);
});

// eine Aufgabe über die ID
app.get("/api/tasks/:id", (req, res) => {
  const task = tasks.find(t => t.id === Number(req.params.id));
  if (!task) return res.status(404).json({ error: "Nicht gefunden" });
  res.json(task);
});

// neue Aufgabe anlegen
app.post("/api/tasks", (req, res) => {
  const { title } = req.body;
  if (!title) return res.status(400).json({ error: "title fehlt" });
  const task = { id: tasks.length + 1, title, done: false };
  tasks.push(task);
  console.log("[api] neue Aufgabe:", task);
  res.status(201).json(task);
});

app.listen(PORT, () => console.log(`[express] http://localhost:${PORT}`));

Vom Browser aus abrufen

// im Frontend (public/app.js)
const response = await fetch("/api/tasks");
const tasks = await response.json();
console.log(tasks);

await fetch("/api/tasks", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ title: "SQL lernen" })
});

Middleware

// läuft bei JEDER Anfrage, bevor die Route drankommt
app.use((req, res, next) => {
  console.log(`[log] ${req.method} ${req.url}`);
  next();                      // weiter zur nächsten Station – nicht vergessen!
});

Tipps & typische Fehler

  • Daten in einem Array gehen beim Neustart verloren – für echte Speicherung eine Datenbank (SQL) oder zumindest eine JSON-Datei nutzen.
  • Eingaben vom Client immer prüfen, bevor du sie verwendest.
  • Zum Testen von APIs ohne Frontend: die Erweiterung „REST Client“ oder „Thunder Client“ in VS Code.
