Code mit require aufteilen und mit Metatabellen Klassen nachbauen.

Module

-- mathutils.lua
local M = {}                          -- Modul ist eine Tabelle

function M.randomInt(min, max)
    return math.random(min, max)
end

return M                              -- am Ende zurückgeben
-- main.lua
local mathutils = require("mathutils")   -- ohne .lua
print(mathutils.randomInt(1, 6))

Klassen mit Metatabellen

local Player = {}
Player.__index = Player              -- fehlende Felder in Player nachschlagen

function Player.new(name)
    local self = setmetatable({}, Player)
    self.name = name
    self.score = 0
    return self
end

function Player:addScore(points)     -- : übergibt automatisch self
    self.score = self.score + points
end

local p = Player.new("Max")
p:addScore(10)                       -- gleich wie Player.addScore(p, 10)
print(p.name, p.score)               --> Max  10
