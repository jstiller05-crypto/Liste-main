C++ · Daten

std::string

Text speichern, zusammensetzen, durchsuchen und umwandeln.

Grundlagen

#include <string>

std::string name = "Max";
std::string greeting = "Hallo " + name + "!";   // zusammensetzen

name.length();        // 3  (auch .size())
name[0];              // 'M' – einzelnes Zeichen (char)
name.empty();         // false
name += " Mustermann";   // anhängen

size_t pos = text.find("C++");
if (pos != std::string::npos) {           // npos = "nicht gefunden"

Ganze Zeilen einlesen

std::string fullName;

std::cout << "Name: ";
std::getline(std::cin, fullName);   // liest bis Enter – mit Leerzeichen

// Achtung nach cin >> zahl: den übrigen Zeilenumbruch erst verwerfen
int age;
std::cin >> age;
std::cin.ignore();                  // '\n' aus dem Puffer entfernen
std::getline(std::cin, fullName);

Umwandeln

int number = std::stoi("42");          // string → int
double value = std::stod("3.14");      // string → double
std::string s = std::to_string(99);    // Zahl → string

// Groß-/Kleinbuchstaben (mit <cctype> und <algorithm>)
std::string word = "Hallo";
std::transform(word.begin(), word.end(), word.begin(), ::toupper);   // "HALLO"

Zeichen durchgehen

std::string input = "a1b2c3";
int digits = 0;

for (char c : input) {
    if (std::isdigit(c)) {   // <cctype>: isdigit, isalpha, isspace, toupper …
        ++digits;
    }
}
std::cout << digits << " Ziffern\n";   // 3

Tipps & typische Fehler

  • "A" ist ein String, 'A' ein einzelnes Zeichen (char).
  • std::stoi("abc") wirft eine Ausnahme – bei Nutzereingaben mit try/catch absichern.
  • cin >> text liest nur bis zum ersten Leerzeichen – für ganze Sätze getline.
