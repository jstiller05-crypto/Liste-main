Bedingungen, Schleifen & Funktionen

if, switch, for-in mit Bereichen und Funktionen mit Argument-Labels.

if & switch

let points = 72

if points >= 90 {                 // keine Klammern um die Bedingung nötig
    print("Sehr gut")
} else if points >= 60 {
    print("Bestanden")
} else {
    print("Nicht bestanden")
}

switch points {
case 90...100: print("Sehr gut")  // Bereiche als Fall
case 60..<90:  print("Bestanden")
default:       print("Nicht bestanden")
}                                  // kein break nötig – kein Durchfallen

switch muss in Swift immer alle Fälle abdecken (notfalls mit default).

Schleifen & Bereiche

for i in 1...5 { print(i) }          // 1 bis 5 (inklusive)
for i in 0..<5 { print(i) }          // 0 bis 4
for i in stride(from: 10, to: 0, by: -2) { print(i) }   // 10 8 6 4 2

let colors = ["rot", "grün", "blau"]
for color in colors { print(color) }
for (index, color) in colors.enumerated() { print(index, color) }

var tries = 0
while tries < 3 { tries += 1 }
repeat { tries -= 1 } while tries > 0

Funktionen

func greet(name: String, greeting: String = "Hallo") -> String {
    return "\(greeting), \(name)!"
}
greet(name: "Max")                    // Argument-Label beim Aufruf

func add(_ a: Int, _ b: Int) -> Int {  // _ = ohne Label aufrufen
    a + b                              // eine Zeile: return optional
}
add(2, 3)

func move(from start: Int, to end: Int) { }   // äußeres + inneres Label
move(from: 1, to: 5)

Argument-Labels machen Aufrufe wie Sätze lesbar: move(from: 1, to: 5).

Closures

let numbers = [4, 9, 1]
let sorted = numbers.sorted { $0 < $1 }          // $0, $1 = erstes/zweites Argument
let doubled = numbers.map { $0 * 2 }
let big = numbers.filter { $0 > 3 }
let sum = numbers.reduce(0, +)

Tipps & typische Fehler

  • ... = inklusive Ende, ..< = ohne Ende.
  • Die Array-Methoden map, filter, reduce, sorted funktionieren wie in JavaScript.
