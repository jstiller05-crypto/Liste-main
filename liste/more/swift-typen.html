Array, Dictionary, Set

var fruits = ["Apfel", "Birne"]
fruits.append("Kiwi")
fruits.remove(at: 0)
fruits.count
fruits.contains("Kiwi")
fruits.first            // Optional!

var scores: [String: Int] = ["Max": 120]
scores["Anna"] = 150
if let maxScore = scores["Max"] { print(maxScore) }   // Zugriff liefert Optional
for (name, score) in scores { print(name, score) }

var used: Set<Int> = [1, 2]
used.insert(2)          // bleibt einmalig

struct & class

struct Task {                       // Werttyp: wird kopiert
    let question: String
    let answer: Int
    var attempts = 0

    mutating func addAttempt() {    // mutating: ändert das struct selbst
        attempts += 1
    }
}

class Player {                      // Referenztyp: wird geteilt
    let name: String
    private(set) var score = 0

    init(name: String) {            // Konstruktor
        self.name = name
    }

    func addScore(_ points: Int) {
        score += points
    }
}

let p = Player(name: "Max")
p.addScore(10)

enum

enum Difficulty {
    case easy, medium, hard
}

enum LoadResult {
    case success(data: String)       // Fall mit angehängtem Wert
    case failure(message: String)
}

let result = LoadResult.failure(message: "Offline")
switch result {
case .success(let data):    print(data)
case .failure(let message): print("[Fehler] \(message)")
}

Protocol

protocol Shape {
    func area() -> Double
}

struct Circle: Shape {
    let radius: Double
    func area() -> Double { .pi * radius * radius }
}

Ein Protocol ist wie ein Interface in C#: Es legt fest, welche Methoden ein Typ haben muss. SwiftUI-Ansichten erfüllen z. B. das Protocol View.

Tipps & typische Fehler

  • Arrays und Dictionaries sind in Swift Werttypen – eine Zuweisung kopiert sie.
  • Mit let deklarierte Structs sind komplett unveränderlich.
